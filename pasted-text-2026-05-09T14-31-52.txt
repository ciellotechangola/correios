2. SUBSTITUIR .single() INCORRETO
==================================================

SE O VENDEDOR PODE NÃO TER LOJA:

EXEMPLO CORRETO:

  .from('lojas')
  .select('*')
  .eq('profile_id', user.id)
  .maybeSingle();

}

}
